cmake_minimum_required(VERSION 3.22)
project(owlwhisp LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(P2 "${REPO}/third_party/phonon-2-c/csrc")
file(GLOB P2_SOURCES "${P2}/*.c")
list(FILTER P2_SOURCES EXCLUDE REGEX "/(main|bench_gemm)\\.c$")
add_library(owlwhisp SHARED engines.cpp ${P2_SOURCES})
target_include_directories(owlwhisp PRIVATE "${P2}" "${REPO}/third_party/voxrt/include" "${REPO}/third_party/needle")
target_compile_options(owlwhisp PRIVATE -O3 -funroll-loops)
target_compile_definitions(owlwhisp PRIVATE _GNU_SOURCE)
target_link_options(owlwhisp PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(owlwhisp PRIVATE m log "${CMAKE_CURRENT_LIST_DIR}/../jniLibs/${ANDROID_ABI}/libvoxrt.so")
if(ANDROID_ABI STREQUAL "arm64-v8a")
    target_compile_definitions(owlwhisp PRIVATE OWL_WHISTLE=1)
    target_link_libraries(owlwhisp PRIVATE "${REPO}/third_party/needle/libneedle.a")
endif()
